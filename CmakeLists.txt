cmake_minimum_required(VERSION 3.14)
project(csv_parser VERSION 1.0)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

#define header-only libary
add_library(csv_parser INTERFACE)

# Set the path to absolute to ensure the compiler can find it reliably
target_include_directories(csv_parser INTERFACE 
    "${CMAKE_CURRENT_SOURCE_DIR}/include"
)

# find catch 2 v3
find_package(Catch2 3 REQUIRED)

# add test execuatable 
add_executable(unit_tests tests/test_main.cpp)

# link unit_tests csv_parser and Catch2
target_link_libraries(unit_tests 
    PRIVATE 
        csv_parser 
        Catch2::Catch2WithMain
)
